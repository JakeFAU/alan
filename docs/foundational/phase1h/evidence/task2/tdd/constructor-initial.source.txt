package adaptive_test

import (
 "testing"
 "github.com/JakeFAU/alan/examples/adaptive"
)

func TestScalarAdaptivePairedResults(t *testing.T) {
 leaf, err := adaptive.ScalarLeaf(true)
 if err != nil { t.Fatal(err) }
 component, err := adaptive.NewScalar(leaf)
 if err != nil { t.Fatal(err) }
 if _, err := adaptive.New(component, false, adaptive.DefaultLimits(1)); err != nil { t.Fatal(err) }
}
