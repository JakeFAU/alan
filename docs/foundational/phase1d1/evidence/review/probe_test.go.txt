func TestReviewStalledBodyAllowsPeer(t *testing.T) {
 ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second); defer cancel()
 held := make(chan struct{}); release := make(chan struct{}); var once sync.Once
 server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  var req resource.Request; if err := json.NewDecoder(r.Body).Decode(&req); err != nil { t.Error(err); return }
  raw, _ := json.Marshal(receiptFor(req)); w.WriteHeader(http.StatusOK); w.Write(raw[:1]); w.(http.Flusher).Flush(); close(held)
  select {case <-release: w.Write(raw[1:]); case <-r.Context().Done():}
 }))
 defer server.Close(); defer once.Do(func(){close(release)})
 p := externalProvision(t,server.URL,"non-idempotent")
 addProgram(t,&p,"peer",[]Instruction{{Op:"end",Outcome:"succeeded"}},map[string]Value{})
 s,_ := messageStore(t,p)
 progressed := false
 ctx = boundary.WithObserver(ctx,func(ctx context.Context,e boundary.Event) error {
  if e.Subject=="peer" && e.Boundary=="execution.staged" { select {case <-held: case <-ctx.Done(): return ctx.Err()} }
  if e.Subject=="peer" && e.Boundary=="execution.committed" { progressed=true; once.Do(func(){close(release)}) }
  return nil
 })
 report,err := s.Run(ctx,RunOptions{MaxSteps:20,MaxDispatch:20,MaxInFlight:1,ExternalTimeout:4*time.Second})
 if err!=nil || !progressed || len(report.UnresolvedIntents)!=0 || current(t,s,"client").Projection.Status!="succeeded" {t.Fatalf("body blocked: %+v %v",report,err)}
 t.Logf("partial body held -> peer commit -> body release -> non-idempotent receipt consumed; report=%+v",report)
}

func TestReviewCloseAccountsAllReceivedCandidates(t *testing.T) {
 ctx,cancel := context.WithTimeout(t.Context(),8*time.Second); defer cancel()
 received:=make(chan string,3); waiting:=make(chan struct{}); var once sync.Once
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){var req resource.Request; json.NewDecoder(r.Body).Decode(&req); json.NewEncoder(w).Encode(receiptFor(req))})); defer server.Close()
 p:=Provision{}
 for i:=0;i<3;i++ {addExternalClient(t,&p,fmt.Sprintf("client%d",i),server.URL,"non-idempotent",int64(i+1))}
 addProgram(t,&p,"zpeer",[]Instruction{{Op:"end",Outcome:"succeeded"}},map[string]Value{})
 s,path:=messageStore(t,p)
 ctx=boundary.WithObserver(ctx,func(ctx context.Context,e boundary.Event) error {
  if e.Boundary=="external.response" { received<-e.Subject; <-ctx.Done(); return ctx.Err() }
  if e.Subject=="zpeer" && e.Boundary=="execution.staged" {once.Do(func(){close(waiting)}); <-ctx.Done(); return ctx.Err()}
  return nil
 })
 type outcome struct {r RunReport; e error}; done:=make(chan outcome,1)
 go func(){r,e:=s.Run(ctx,RunOptions{MaxSteps:30,MaxDispatch:30,MaxInFlight:3}); done<-outcome{r,e}}()
 for i:=0;i<3;i++ {select{case <-received: case <-ctx.Done():t.Fatal(ctx.Err())}}
 select{case <-waiting: case <-ctx.Done():t.Fatal(ctx.Err())}
 if err:=s.Close();err!=nil {t.Fatal(err)}
 got:=<-done
 if !errors.Is(got.e,context.Canceled)||len(got.r.UnknownExternal)!=0||len(got.r.UnresolvedIntents)!=0||got.r.Dispatches!=3 {t.Fatalf("Close lost candidates: %+v %v",got.r,got.e)}
 reopened,err:=Open(t.Context(),path,DefaultLimits());if err!=nil {t.Fatal(err)};defer reopened.Close()
 report,err:=reopened.Run(t.Context(),RunOptions{MaxSteps:30,MaxDispatch:1})
 if err!=nil||report.Dispatches!=0||count(t,reopened,"external_receipts")!=3 {t.Fatalf("recovery: %+v %v",report,err)}
 for i:=0;i<3;i++ {if current(t,reopened,fmt.Sprintf("client%d",i)).Projection.Status!="succeeded"{t.Fatal("unconsumed")}}
 t.Logf("3 valid received candidates -> concurrent Close -> joined/persisted -> reopen consumed, without redispatch; report=%+v",report)
}
