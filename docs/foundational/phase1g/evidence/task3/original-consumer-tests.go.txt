// Literal expectations are frozen before the implementation; no compiler,
// reducer, replay or collection helper calculates expected item correspondence.
var ids = []string{"item-a","item-b","item-c","item-d","item-e","item-f","item-g","item-h","item-i"}
var inputs = []int64{2,-3,2,0,11,-1,9,4,100}
var outputs = []int64{4,-6,4,0,22,-2,18,8}

func work(n int) schema.Value {
 rows:=make([]map[string]schema.Value,n)
 for i:=range rows { rows[i]=map[string]schema.Value{"item_id":schema.Text(ids[i]),"number":schema.Int(inputs[i])} }
 return schema.List(rows...)
}

// This fails if the component serializes/rebinds handles, coalesces equal
// payloads, changes retained topology, accepts invalid work or loses ordering.
func TestRetainedCollectionContract(t *testing.T) {
 leaf,err:=worktree.Deterministic(); if err!=nil {t.Fatal(err)}
 collection,err:=compose.New(leaf,compose.DefaultOptions()); if err!=nil {t.Fatal(err)}
 in,out,err:=collection.Schemas(); if err!=nil {t.Fatal(err)}
 for _,nested:=range []bool{false,true} {
  deployment,err:=worktree.New(leaf,nested,worktree.DefaultLimits(2)); if err!=nil {t.Fatal(err)}
  if deployment.Collection.ID()!=collection.ID() {t.Fatal("different collection template")}
  di,do,err:=deployment.Collection.Schemas(); if err!=nil||di.ID!=in.ID||do.ID!=out.ID {t.Fatal("schema drift",err)}
  before:=deployment.Collection.Bytes()
  for _,n:=range []int{0,1,3,7,8,9} {t.Run(fmt.Sprintf("nested=%v/n=%d",nested,n),func(t *testing.T){
   path:=filepath.Join(t.TempDir(),"work.sqlite")
   s,err:=runtime.Create(t.Context(),path,deployment.Provision,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}
   defer s.Close()
   initial,err:=runtime.Inspect(t.Context(),path,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}
   if len(initial.Tables["runs"])!=1 {t.Fatal("work expanded at initialization")}
   admitted,err:=worktree.Submit(t.Context(),s,"work",work(n)); if err!=nil {t.Fatal(err)}
   if n==9 { if admitted.Accepted {t.Fatal("ninth item accepted")}; if len(initial.Tables["composition_invocations"])!=0 {t.Fatal("children before input")}; return }
   if !admitted.Accepted {t.Fatal(admitted)}
   report,err:=s.Run(t.Context(),runtime.RunOptions{MaxSteps:2000,MaxDispatch:500,MaxInFlight:2}); if err!=nil {t.Fatal(err)}
   if len(report.Roots)!=1||report.Roots[0].Outcome==nil||report.Roots[0].Outcome.Status!="succeeded" {t.Fatalf("root incomplete: %+v",report)}
   state,err:=s.Snapshot(t.Context(),runtime.RunID{Subject:"root",Generation:1}); if err!=nil {t.Fatal(err)}
   rows,err:=state.Projection.Data["results"].Records(); if err!=nil||len(rows)!=n {t.Fatal(rows,err)}
   for i,row:=range rows {
    response,err:=row["responses"].Records(); if err!=nil||len(response)!=1 {t.Fatal(row,err)}
    if row["item_id"].Text!=ids[i]||row["status"].Text!="succeeded"||response[0]["doubled"].Number!=outputs[i]||row["operation"].Text=="" {t.Fatal("incorrect item mapping",i,row)}
   }
   enclosing:=int64(1); if nested {enclosing=2}
   if report.Roots[0].Charges.Invocations!=int64(n)+enclosing||report.Roots[0].Charges.CreatedRuns!=int64(n)+enclosing||report.Roots[0].Charges.Effects!=0 {t.Fatal("shared charge mismatch",report.Roots)}
   final,err:=runtime.Inspect(t.Context(),path,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}
   if len(final.Tables["programs"])!=len(initial.Tables["programs"])||len(final.Tables["composition_invocations"])!=n+int(enclosing) {t.Fatal("topology or invocation count mismatch")}
   if !bytes.Equal(before,deployment.Collection.Bytes()) {t.Fatal("artifact changed after input")}
   if _,err=runtime.Replay(t.Context(),path,runtime.DefaultLimits()); err!=nil {t.Fatal(err)}
  })}
 }
}

func TestRejectDuplicateIDsBeforeAnyChild(t *testing.T) {
 leaf,err:=worktree.Deterministic(); if err!=nil {t.Fatal(err)}
 d,err:=worktree.New(leaf,false,worktree.DefaultLimits(1)); if err!=nil {t.Fatal(err)}
 path:=filepath.Join(t.TempDir(),"work.sqlite"); s,err:=runtime.Create(t.Context(),path,d.Provision,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}; defer s.Close()
 v:=schema.List(map[string]schema.Value{"item_id":schema.Text("item-a"),"number":schema.Int(2)},map[string]schema.Value{"item_id":schema.Text("item-a"),"number":schema.Int(-3)})
 a,err:=worktree.Submit(t.Context(),s,"duplicate",v); if err!=nil {t.Fatal(err)}; if a.Accepted {t.Fatal("duplicate IDs accepted")}
 snapshot,err:=runtime.Inspect(t.Context(),path,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}; if len(snapshot.Tables["composition_invocations"])!=0 {t.Fatal("rejected work created children")}
}

func TestLowerBudgetPreservesAcceptedWorkAndOrderedRejections(t *testing.T) {
 for _,field:=range []string{"invocations","runs","depth"} {t.Run(field,func(t *testing.T){
  leaf,err:=worktree.Deterministic(); if err!=nil {t.Fatal(err)}
  limits:=worktree.DefaultLimits(1)
  switch field {case "invocations":limits.MaxInvocations=2;case "runs":limits.MaxCreatedRuns=2;case "depth":limits.MaxDepth=1}
  d,err:=worktree.New(leaf,false,limits); if err!=nil {t.Fatal(err)}
  path:=filepath.Join(t.TempDir(),"work.sqlite"); s,err:=runtime.Create(t.Context(),path,d.Provision,runtime.DefaultLimits()); if err!=nil {t.Fatal(err)}; defer s.Close()
  a,err:=worktree.Submit(t.Context(),s,"work",work(3)); if err!=nil||!a.Accepted {t.Fatal(a,err)}
  report,err:=s.Run(t.Context(),runtime.RunOptions{MaxSteps:2000,MaxDispatch:500,MaxInFlight:1}); if err!=nil {t.Fatal(err)}
  if report.Roots[0].Outcome==nil||report.Roots[0].Outcome.Status!="failed" {t.Fatal("budget rejection reported success",report)}
  state,err:=s.Snapshot(t.Context(),runtime.RunID{Subject:"root",Generation:1}); if err!=nil {t.Fatal(err)}
  rows,err:=state.Projection.Data["results"].Records(); if err!=nil||len(rows)!=3 {t.Fatal("lost accepted work",rows,err)}
  for i,row:=range rows {if row["item_id"].Text!=ids[i] {t.Fatal("lost ordering",row)}; want:="rejected";if i==0&&field!="depth" {want="succeeded"};if row["status"].Text!=want {t.Fatal(field,row)}}
  if _,err=runtime.Replay(t.Context(),path,runtime.DefaultLimits()); err!=nil {t.Fatal(err)}
 })}
}

func TestPublicSchemaCopiesAndReservedField(t *testing.T) {
 if _,_,err:=(author.Template{}).Schemas(); err==nil {t.Fatal("zero template schemas")}
 leaf,err:=worktree.Deterministic(); if err!=nil {t.Fatal(err)}
 in,out,err:=leaf.Schemas(); if err!=nil {t.Fatal(err)}
 in.Record["number"]=schema.Field{Kind:"bool"};out.Record["doubled"]=schema.Field{Kind:"bool"}
 again,result,err:=leaf.Schemas(); if err!=nil||again.Record["number"].Kind!="int"||result.Record["doubled"].Kind!="int" {t.Fatal("mutable retained schemas",err)}
 a,err:=author.Compile(author.Definition{Ports:map[string]author.Port{"public":{Peer:"caller",Obligation:"work",Receive:schema.Record{"item_id":{Kind:"string",MaxBytes:20}}}},Steps:[]author.Step{author.Receive("public",map[string]string{"item_id":"id"}),author.Send("public",nil),author.Success()}});if err!=nil {t.Fatal(err)}
 conflict,err:=author.NewTemplate(author.Application{"entry":{Artifact:a}},"entry","public");if err!=nil {t.Fatal(err)}
 if _,err=compose.New(conflict,compose.DefaultOptions());err==nil {t.Fatal("reserved item_id conflict accepted")}
}
