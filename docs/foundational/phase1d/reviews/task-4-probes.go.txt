package executor
import ("testing";"reflect")
func TestReviewExternalDeniedGrantAndReceiptTransportNamespace(t *testing.T) {
 p:=externalProvision(t,"http://127.0.0.1:1/apply","idempotent")
 p.Permissions=[]Permission{{Kind:"input",Source:"local",Target:"client",Operation:"input",Revision:0}}
 s,path:=messageStore(t,p)
 state,err:=s.Commit(t.Context(),current(t,s,"client").Authority(),"");if err!=nil{t.Fatal(err)}
 id:=state.Projection.Data["pending"].Text
 in:=Input{TransportID:identity("receipt-delivery",id),LogicalID:id,Target:"client",Generation:1,Obligation:"receipt",Operation:id,Payload:map[string]Value{"total":Int(999),"accepted":{Kind:"bool",Flag:true}}}
 admission,err:=s.AdmitInput(t.Context(),in);if err!=nil||!admission.Accepted{t.Fatal(admission,err)}
 if _,err=s.Run(t.Context(),RunOptions{100,100});err!=nil{t.Fatal(err)}
 o,err:=s.externalOutcome(t.Context(),id);if err!=nil||o.Disposition!="grant_denied"||o.Attempt.Allowed{t.Fatal(o,err)}
 auth,err:=loadAuthorization(t.Context(),s.conn,id);if err!=nil||!auth.Allowed{t.Fatal(auth,err)}
 if current(t,s,"client").Projection.Data["total"]!=Int(0)||current(t,s,"client").Projection.Status!="succeeded"{t.Fatal("forged input consumed as receipt")}
 before,err:=Inspect(t.Context(),path,DefaultLimits());if err!=nil{t.Fatal(err)}
 s.Close()
 if _,err=Replay(t.Context(),path,DefaultLimits());err!=nil{t.Fatal(err)}
 after,err:=Inspect(t.Context(),path,DefaultLimits());if err!=nil||!reflect.DeepEqual(before,after){t.Fatal("replay mutated",err)}
 s,err=Open(t.Context(),path,DefaultLimits());if err!=nil{t.Fatal(err)};defer s.Close()
 again,err:=s.AdmitInput(t.Context(),in);if err!=nil||!reflect.DeepEqual(admission,again){t.Fatal("retained raw input changed",again,err)}
 if settled,err:=s.DispatchExternal(t.Context(),id);err!=nil||!settled{t.Fatal(settled,err)}
 t.Log("source authorization allowed, separate external grant denied; raw receipt transport input retained without forged receipt; reopen/replay exact")
}
