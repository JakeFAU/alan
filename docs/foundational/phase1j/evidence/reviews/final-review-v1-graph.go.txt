package main
import (
 "fmt"
 "encoding/json"
 "os"
 "math/big"
 j "github.com/JakeFAU/alan/research/join"
 s "github.com/JakeFAU/alan/research/substitution"
)
type counts struct{Variant s.Variant; Configurations,States,Edges,Maximal,Success,Pending,Visited,MinLongest,MaxLongest int}
func main(){
 for _,v:=range []s.Variant{s.Reference,s.Local,s.Helper}{
 total:=counts{Variant:v,MinLongest:1000000}
 for _,c:=range j.Configurations(){
 states:=[]j.State{j.Initial(c)}; seen:=map[j.State]int{states[0]:0}; next:=[][]int{{}}; indegree:=[]int{0}; edgeCount:=0
 for i:=0;i<len(states);i++ {for _,e:=range j.Successors(c,v,states[i]){
  k,ok:=seen[e.Next]; if !ok{k=len(states); seen[e.Next]=k;states=append(states,e.Next);next=append(next,nil);indegree=append(indegree,0)}
  next[i]=append(next[i],k);indegree[k]++;edgeCount++
 }}
 queue:=[]int{}; depth:=make([]int,len(states)); paths:=make([]big.Int,len(states));paths[0].SetInt64(1)
 for i,n:=range indegree{if n==0{queue=append(queue,i)}}
 var maximalPaths big.Int; successes,pending,maximal,longest:=0,0,0,0
 for head:=0;head<len(queue);head++{i:=queue[head];x:=states[i]
  if len(next[i])==0{maximal++;maximalPaths.Add(&maximalPaths,&paths[i]);if depth[i]>longest{longest=depth[i]};switch x.Aggregate{case "succeeded":successes++;case "":if x.ReportStage!=3||x.ReportAllowed||x.ReportDelivered||x.Base.RootStatus!=""||x.Base.PeerStatus!="succeeded"||!x.Base.Ops[0].Consumed||!x.Base.Ops[3].Consumed{panic("non-witness pending maximum")};pending++;default:panic("unexpected aggregate")}}
  for _,k:=range next[i]{indegree[k]--;paths[k].Add(&paths[k],&paths[i]);if depth[i]+1>depth[k]{depth[k]=depth[i]+1};if indegree[k]==0{queue=append(queue,k)}}
 }
 if len(queue)!=len(states){panic("cycle independently found")}
 reference:=j.Explore(c,v)
 if len(states)!=reference.States||edgeCount!=reference.Transitions||maximal!=reference.MaximalStates||maximalPaths.String()!=reference.MaximalExecutions{panic("independent count differs")}
 total.Configurations++;total.States+=len(states);total.Edges+=edgeCount;total.Maximal+=maximal;total.Success+=successes;total.Pending+=pending;total.Visited+=len(queue)
 if longest<total.MinLongest{total.MinLongest=longest};if longest>total.MaxLongest{total.MaxLongest=longest}
 }
 b,_:=json.Marshal(total);fmt.Fprintln(os.Stdout,string(b))
 }
}
